# Subtasks Implementation Plan

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** Enable hierarchical task relationships where a parent task can have child tasks (subtasks) that are full-featured tasks with their own dates, priorities, tags, and dependencies.

**Architecture:** Add `parent_task_id` column to tasks table for self-referential hierarchy. Single-level nesting only (no grandchildren). Block parent completion until all children complete. Inherit priority and tags on creation, then independent.

**Tech Stack:** Go backend with PostgreSQL migrations, React/TypeScript frontend, REST API endpoints.

**Spec:** See `docs/subtasks-spec.md` for full design document.

```go
// prepareSubtask creates a subtask from input, inheriting parent properties
func prepareSubtask(parent *models.Task, input CreateTaskInput) *models.Task {
	subtask := &models.Task{
		ParentTaskID: &parent.ID,
		Title:        input.Title,
		Description:  input.Description,
		Priority:     input.Priority,
		Status:       input.Status,
	}

	// Inherit priority from parent if not provided
	if subtask.Priority == nil && parent.Priority != nil {
		subtask.Priority = parent.Priority
	}

	// Inherit tags from parent
	for _, tag := range parent.Tags {
		subtask.Tags = append(subtask.Tags, tag)
	}

	return subtask
}
```

```go
import "errors"

// validateParentAssignment checks if a task can be assigned to a parent
func validateParentAssignment(task *models.Task, parent *models.Task) error {
	// No self-reference
	if task.ID == parent.ID {
		return errors.New("task cannot be its own parent")
	}

	// Single level only: parent cannot have a parent
	if parent.ParentTaskID != nil {
		return errors.New("cannot nest more than one level deep")
	}

	// Task with children cannot become a subtask
	if len(task.Subtasks) > 0 {
		return errors.New("cannot make a parent task into a subtask")
	}

	return nil
}
```

```go
// validateTaskCompletion checks if a task can be completed
func validateTaskCompletion(task *models.Task, force bool) error {
	if force {
		return nil
	}

	incompleteCount := 0
	for _, subtask := range task.Subtasks {
		if !subtask.IsComplete {
			incompleteCount++
		}
	}

	if incompleteCount > 0 {
		return &IncompleteSubtaskError{
			IncompleteCount: incompleteCount,
			TotalSubtasks:   len(task.Subtasks),
		}
	}

	return nil
}
```

```go
// HandleCreateSubtask creates a new task as a subtask of a parent
func HandleCreateSubtask(w http.ResponseWriter, r *http.Request) {
	userID := getUserID(r)
	if userID == 0 {
		http.Error(w, "Unauthorized", http.StatusUnauthorized)
		return
	}

	// Get parent task ID from URL
	vars := mux.Vars(r)
	parentID, err := strconv.Atoi(vars["id"])
	if err != nil {
		http.Error(w, "Invalid parent task ID", http.StatusBadRequest)
		return
	}

	// Parse request body
	var input services.CreateTaskInput
	if err := json.NewDecoder(r.Body).Decode(&input); err != nil {
		http.Error(w, "Invalid request body", http.StatusBadRequest)
		return
	}

	// Get parent task
	parent, err := services.GetTaskByID(s.GetDB(), userID, parentID)
	if err != nil {
		http.Error(w, "Parent task not found", http.StatusNotFound)
		return
	}

	// Prepare subtask with inheritance
	subtask := services.PrepareSubtask(parent, input)

	// Create the subtask
	createdTask, err := services.CreateTask(s.GetDB(), userID, subtask)
	if err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	// Add tags to subtask
	if len(subtask.Tags) > 0 {
		services.AddTagsFromTask(userID, createdTask.ID)
	}

	w.Header().Set("Content-Type", "application/json")
	w.WriteHeader(http.StatusCreated)
	json.NewEncoder(w).Encode(createdTask)
}
```

```go
// HandleSetTaskParent sets or clears the parent of a task
func HandleSetTaskParent(w http.ResponseWriter, r *http.Request) {
	userID := getUserID(r)
	if userID == 0 {
		http.Error(w, "Unauthorized", http.StatusUnauthorized)
		return
	}

	// Get task ID from URL
	vars := mux.Vars(r)
	taskID, err := strconv.Atoi(vars["id"])
	if err != nil {
		http.Error(w, "Invalid task ID", http.StatusBadRequest)
		return
	}

	// Parse request body
	var input struct {
		ParentTaskID *int `json:"parent_task_id"`
	}
	if err := json.NewDecoder(r.Body).Decode(&input); err != nil {
		http.Error(w, "Invalid request body", http.StatusBadRequest)
		return
	}

	// Get the task
	task, err := services.GetTaskByID(s.GetDB(), userID, taskID)
	if err != nil {
		http.Error(w, "Task not found", http.StatusNotFound)
		return
	}

	// If setting a parent (not clearing)
	if input.ParentTaskID != nil {
		// Get parent task
		parent, err := services.GetTaskByID(s.GetDB(), userID, *input.ParentTaskID)
		if err != nil {
			http.Error(w, "Parent task not found", http.StatusNotFound)
			return
		}

		// Load subtasks for validation
		task.Subtasks, _ = services.GetSubtasks(s.GetDB(), userID, taskID)

		// Validate the assignment
		if err := services.ValidateParentAssignment(task, parent); err != nil {
			http.Error(w, err.Error(), http.StatusBadRequest)
			return
		}
	}

	// Update the task
	task.ParentTaskID = input.ParentTaskID
	if err := services.UpdateTaskParent(s.GetDB(), userID, taskID, input.ParentTaskID); err != nil {
		http.Error(w, err.Error(), http.StatusInternalServerError)
		return
	}

	// Return updated task
	updatedTask, _ := services.GetTaskByID(s.GetDB(), userID, taskID)
	w.Header().Set("Content-Type", "application/json")
	json.NewEncoder(w).Encode(updatedTask)
}
```

```go
// ValidateParentAssignment is the exported version for handlers
func ValidateParentAssignment(task *models.Task, parent *models.Task) error {

}

// GetSubtasks retrieves all subtasks for a parent task
func GetSubtasks(db *sql.DB, userID int, parentTaskID int) ([]models.Task, error) {
	query := `
		SELECT id, title, is_complete, status, priority, scheduled_date, due_date, created_at, updated_at
		FROM tasks
		WHERE user_id = $1 AND parent_task_id = $2 AND is_deleted = FALSE
		ORDER BY created_at
	`

	rows, err := db.Query(query, userID, parentTaskID)
	if err != nil {
		return nil, err
	}
	defer rows.Close()

	var subtasks []models.Task
	for rows.Next() {
		var t models.Task
		err := rows.Scan(&t.ID, &t.Title, &t.IsComplete, &t.Status, &t.Priority, &t.ScheduledDate, &t.DueDate, &t.CreatedAt, &t.UpdatedAt)
		if err != nil {
			return nil, err
		}
		subtasks = append(subtasks, t)
	}

	return subtasks, nil
}

// UpdateTaskParent updates the parent_task_id of a task
func UpdateTaskParent(db *sql.DB, userID int, taskID int, parentTaskID *int) error {
	query := `UPDATE tasks SET parent_task_id = $1, updated_at = NOW() WHERE id = $2 AND user_id = $3`
	_, err := db.Exec(query, parentTaskID, taskID, userID)
	return err
}
```

```typescript
import { Task } from "../../models/Task";
```

```typescript
interface TaskListProps {
  tasks: Task[];
  onTagClick: (tag: string) => void;
  hideMatrixTags?: boolean;
  selectMode?: boolean;
  selectedTaskIds?: Set<number>;
  onTaskSelect?: (taskId: number) => void;
  onTaskClick?: (taskId: number) => void;
}

export function TaskList({
  tasks,
  onTagClick,
  hideMatrixTags = false,
  selectMode = false,
  selectedTaskIds = new Set(),
  onTaskSelect,
  onTaskClick,
}: TaskListProps) {
  // Separate root tasks from subtasks
  const { rootTasks, subtasksByParent } = useMemo(() => {
    const rootTasks: Task[] = [];
    const subtasksByParent: Record<number, Task[]> = {};

    tasks.forEach((task) => {
      if (task.parent_task_id) {
        // This is a subtask
        if (!subtasksByParent[task.parent_task_id]) {
          subtasksByParent[task.parent_task_id] = [];
        }
        subtasksByParent[task.parent_task_id].push(task);
      } else {
        // This is a root task
        rootTasks.push(task);
      }
    });

    return { rootTasks, subtasksByParent };
  }, [tasks]);

  return (
    <ul className="divide-y divide-slate-200">
      {rootTasks.map((task) => {
        const subtasks = subtasksByParent[task.id] || [];

        if (subtasks.length > 0) {
          // Render as nested group
          return (
            <li key={task.id} className="py-1">
              <TaskNestedGroup
                task={{ ...task, subtasks }}
                onTagClick={onTagClick}
                onTaskClick={onTaskClick || (() => {})}
                selectMode={selectMode}
                selectedTaskIds={selectedTaskIds}
                onTaskSelect={onTaskSelect}
              />
            </li>
          );
        }

        // Render as single item
        return (
          <li key={task.id} className="py-1">
            <TaskListItem
              task={task}
              onTagClick={onTagClick}
              hideMatrixTags={hideMatrixTags}
              selectMode={selectMode}
              isSelected={selectedTaskIds.has(task.id)}
              onSelect={() => onTaskSelect?.(task.id)}
              onClick={() => onTaskClick?.(task.id)}
            />
          </li>

    </ul>
```

```markdown
# Subtasks Feature

## Overview
Tasks can have child tasks (subtasks) for breaking down complex work.

## Creating Subtasks
1. Open a task
2. Click "+ Add Subtask" in the subtasks section
3. Enter the subtask title
4. Press Enter to create

## Behavior
- Subtasks inherit parent's priority and tags on creation
- After creation, subtasks are independent (changes don't affect each other)
- Completing a parent requires all subtasks to be complete first
- Use "Force Complete" to bypass this restriction

## Views
- **List View**: Subtasks nested under parent with collapse/expand
- **Kanban View**: Subtasks shown under parent card
- **Calendar View**: Subtasks appear on their scheduled dates

## Keyboard Shortcuts
- Click parent to toggle collapse/expand
- Standard task shortcuts apply to subtasks
```
